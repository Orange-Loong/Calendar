## 二、认证与授权

### 1. JWT认证

```javascript
// JWT生成和验证配置
const jwt = require('jsonwebtoken');

// 生成JWT令牌
const generateToken = (user) => {
  return jwt.sign(
    { id: user._id, email: user.email },
    process.env.JWT_SECRET,
    { expiresIn: '7d' } // 令牌有效期7天
  );
};

// 验证JWT令牌
const verifyToken = (token) => {
  try {
    return jwt.verify(token, process.env.JWT_SECRET);
  } catch (error) {
    return null;
  }
};
```

### 2. 密码管理

```javascript
// 密码哈希和验证
const bcrypt = require('bcrypt');

// 哈希密码
const hashPassword = async (password) => {
  const saltRounds = 12; // 高强度哈希
  return await bcrypt.hash(password, saltRounds);
};

// 验证密码
const verifyPassword = async (password, hashedPassword) => {
  return await bcrypt.compare(password, hashedPassword);
};
```

### 2. 请求验证

```javascript
// 使用Joi进行请求验证
const Joi = require('joi');

// 日程创建验证规则
const eventSchema = Joi.object({
  title: Joi.string().required().min(1).max(255),
  startTime: Joi.date().required(),
  endTime: Joi.date().required().greater(Joi.ref('startTime')),
  location: Joi.string().max(255),
  description: Joi.string().max(1000)
});

// 任务审批验证规则
const taskApproveSchema = Joi.object({
  approved: Joi.boolean().required(),
  feedback: Joi.string().max(500)
});

// 积分兑换验证规则
const pointsRedeemSchema = Joi.object({
  amount: Joi.number().required().min(1),
  optionId: Joi.string().required()
});

// 验证中间件
const validateRequest = (schema) => {
  return (req, res, next) => {
    const { error } = schema.validate(req.body);
    if (error) {
      return res.status(400).json({ error: error.details[0].message });
    }
    next();
  };
};
```

### 3. 速率限制

```javascript
// 使用express-rate-limit实现速率限制
const rateLimit = require('express-rate-limit');

// API速率限制
const apiLimiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15分钟
  max: 100, // 每个IP最多100个请求
  standardHeaders: true,
  legacyHeaders: false,
  message: { error: 'Too many requests, please try again later.' }
});

// 登录速率限制（更严格）
const loginLimiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15分钟
  max: 10, // 每个IP最多10个登录请求
  standardHeaders: true,
  legacyHeaders: false,
  message: { error: 'Too many login attempts, please try again later.' }
});
```

```javascript
// React Native中使用安全存储
import * as SecureStore from 'expo-secure-store';

// 存储敏感数据
const storeData = async (key, value) => {
  try {
    await SecureStore.setItemAsync(key, value);
  } catch (error) {
    console.error('Failed to store data securely:', error);
  }
};

// 获取敏感数据
const getData = async (key) => {
  try {
    return await SecureStore.getItemAsync(key);
  } catch (error) {
    console.error('Failed to get data securely:', error);
    return null;
  }
};
```
